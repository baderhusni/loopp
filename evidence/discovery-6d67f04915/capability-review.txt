  inputs   : member_id:string
  outputs  : member_name:string[pii], savings_balance:number[internal], savings_account_number:string[pii]
  risk     : max=safe approval=draft

    - enter_operator_id: enter the service operator id [type into textbox "Operator ID" = <secret:core_operator_id>]
